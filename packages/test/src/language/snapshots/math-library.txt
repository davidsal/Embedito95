// LSP view of the `math` core library (hover types + diagnostics).
// Each case is a library-style program (isLibrary) whose probes are
// exported, so unused-declaration noise is suppressed. Types come from
// resolveProjectTypes (the same inference the language server uses for
// hover), reading the real core/math/main.e95 through
// collectLibraryModuleInfo. Diagnostics come from the shared check().
// integer and bit helpers infer their declared types
diagnostics:
- none
types:
- integerResult => i32
- clamped => i32
- smallest => i32
- bitResult => u32

// integer arguments select the width that holds them
diagnostics:
- none
types:
- wideClamped => i64
- narrowClamped => i32
- wideMagnitude => u32
- wideMinimum => i64
- wideBoundClamped => i64
- wideBoundRemapped => i64

// float helpers infer f32
diagnostics:
- none
types:
- remapped => f32
- root => f32
- rounded => f32
- radians => f32

// bit read infers boolean
diagnostics:
- none
types:
- flag => boolean

// inverse trigonometry, logarithms and exponentials infer f32
diagnostics:
- none
types:
- angle => f32
- arc => f32
- natural => f32
- decimal => f32
- raised => f32

// square, remainder and truncate infer f32
diagnostics:
- none
types:

- whole => f32

// f64 arguments select the f64 overload
diagnostics:
- none

// a mixed-width call converts silently with a hover-only note

// an f64 variable with integer literals keeps the f64 overload

- none
types:
- wholeClamp => f64
- wholeRemapped => f64

// byte helpers infer u8 and bitMask infers u32
