// Numeric conversion rule for arguments/constants/let and overload
// resolution by parameter types. Numeric conversions are implicit (auto cast
// with truncation); none of them is reported as a diagnostic, the hover
// explains the possible loss.
// an f64 argument keeps the f64 overload and emits no warning
diagnostics:
- none
types:
- scaled => f64

// an integer literal resolves the f32 overload
diagnostics:
- none
types:
- scaled => f32

// f64 into an f32 context converts silently; the hover explains the precision loss
diagnostics:
- none
types:
- scaled => f32

// u32 * i16 does not fit f32 and selects the f64 overload
diagnostics:
- none
types:
- scaled => f64

// u64 into an i32 parameter converts silently but keeps the declared type
diagnostics:
- none
types:
- narrowed => i32

// i32 into an f32 parameter converts silently; the hover explains the range loss
diagnostics:
- none
types:
- converted => f32

// i8 into an f32 parameter fits exactly and stays silent
diagnostics:
- none
types:
- converted => f32
